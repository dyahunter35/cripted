1. النشأة وفلسفة التشفير التدفقي

صمم عالم التشفير الشهير Ron Rivest خوارزمية RC4 عام 1987 لصالح شركة RSA Security. ظلت الخوارزمية سراً تجارياً حتى عام 1994 عندما نُشرت تفاصيلها مجهولة المصدر على القائمة البريدية Cypherpunks.

على عكس خوارزميات الكتل (مثل DES و GOST) التي تقسم البيانات إلى مجموعات من 8 أو 16 بايت، فإن التشفير التدفقي يعالج البيانات بايتاً ببايت أو بتاً ببت عبر توليد تيار من البايتات شبه العشوائية (Keystream) ثم دمجه مع النص الأصلي عبر بوابة XOR البسيطة.

⚙️

2. خوارزميتي العمل الأساسيتين: KSA و PRGA

تحتوي الخوارزمية داخلياً على مصفوفة تباديل $S$ بحجم 256 بايت ($0 \dots 255$) ومؤشرين $i, j$:

خوارزمية جدولة المفتاح (KSA)

تهيئة وخلط المصفوفة $S$ بالاعتماد على المفتاح السري:

  1. تهيئة المصفوفة: S[i] = i لكل قيم $i$ من 0 إلى 255.
  2. ضبط المؤشر j = 0.
  3. خلط العناصر عبر 256 تكراراً:
    j = (j + S[i] + Key[i mod key_len]) mod 256
    تبديل swap(S[i], S[j]).

خوارزمية توليد التيار العشوائي (PRGA)

توليد بايت تيار المفتاح لكل بايت من النص المراد تشفيره:

  1. i = (i + 1) mod 256
  2. j = (j + S[i]) mod 256
  3. تبديل swap(S[i], S[j])
  4. استخراج بايت المفتاح: K = S[(S[i] + S[j]) mod 256]
  5. التشفير / فك التشفير: CipherByte = PlainByte ⊕ K
⚠️

3. الثغرات الأمنية وسبب الإيقاف الرسمي

شكلت RC4 لسنوات العمود الفقري لأمن الإنترنت (في أنظمة SSL/TLS وشبكات Wi-Fi WEP)، لكن الدراسات الأكاديمية كشفت انحيازات إحصائية خطيرة (Biases) في البايتات الأولى من تيار المفاتيح:

🚫
أبرز الهجمات المكتشفة:
  • هجوم FMS (Fluhrer, Mantin, Shamir 2001): أدى إلى كسر مفاتيح WEP في دقائق بسبب ضعف دمج ناقل التهيئة (IV) مع المفتاح.
  • انحياز البايت الثاني (Mantin-Shamir Bias): احتمالية ظهور الصفر في البايت الثاني تبلغ ضعف الاحتمالية العادية (1/128 بدلاً من 1/256).
  • حظر المعيار في TLS (RFC 7465 عام 2015): حظرت هيئة معايير الإنترنت IETF استخدام جميع مجموعات تشفير RC4 في بروتوكولات TLS نهائياً.

الكود اليدوي الكامل — rc4/rc4_manual.py 
"""RC4 (Rivest Cipher 4) implemented from scratch + steganography."""

import argparse
import hashlib
import math
import os
import struct
import wave

from stego_image import hide as img_hide, extract as img_extract
from stego_audio import hide as aud_hide, extract as aud_extract


def _ksa(key):
    key = list(key)
    S = list(range(256))
    j = 0
    for i in range(256):
        j = (j + S[i] + key[i % len(key)]) % 256
        S[i], S[j] = S[j], S[i]
    return S


def _prga(S, n):
    i = j = 0
    out = bytearray()
    for _ in range(n):
        i = (i + 1) % 256
        j = (j + S[i]) % 256
        S[i], S[j] = S[j], S[i]
        out.append(S[(S[i] + S[j]) % 256])
    return out


def _prga_drop(S, n, drop=768):
    i = j = 0
    # Discard initial keystream bytes to eliminate biases (RC4-drop[n])
    for _ in range(drop):
        i = (i + 1) % 256
        j = (j + S[i]) % 256
        S[i], S[j] = S[j], S[i]
    out = bytearray()
    for _ in range(n):
        i = (i + 1) % 256
        j = (j + S[i]) % 256
        S[i], S[j] = S[j], S[i]
        out.append(S[(S[i] + S[j]) % 256])
    return out


def derive_key(password):
    return hashlib.sha256(password.encode()).digest()


def _rc4(data, password):
    key = derive_key(password)
    data = data.encode() if isinstance(data, str) else data
    ks = _prga(_ksa(key), len(data))
    return bytes(b ^ k for b, k in zip(data, ks))


def rc4_encrypt(plaintext, password):
    return _rc4(plaintext, password)


def rc4_decrypt(ciphertext, password):
    return _rc4(ciphertext, password)


def rc4_drop_encrypt(plaintext, password, drop=768):
    key = derive_key(password)
    data = plaintext.encode() if isinstance(plaintext, str) else plaintext
    ks = _prga_drop(_ksa(key), len(data), drop)
    return bytes(b ^ k for b, k in zip(data, ks))


def rc4_drop_decrypt(ciphertext, password, drop=768):
    return rc4_drop_encrypt(ciphertext, password, drop)


def make_sample_image(path, w=64, h=64):
    row_size = (w * 3 + 3) & ~3
    pixel_offset = 54
    file_size = pixel_offset + row_size * h
    header = bytearray(54)
    header[0:2] = b"BM"
    struct.pack_into("<I", header, 2, file_size)
    struct.pack_into("<I", header, 10, pixel_offset)
    struct.pack_into("<I", header, 14, 40)
    struct.pack_into("<i", header, 18, w)
    struct.pack_into("<i", header, 22, h)
    struct.pack_into("<H", header, 26, 1)
    struct.pack_into("<H", header, 28, 24)
    pixels = bytearray(row_size * h)
    for y in range(h):
        for x in range(w):
            idx = (h - 1 - y) * row_size + x * 3
            pixels[idx] = (x * 4) & 0xFF
            pixels[idx + 1] = (y * 4) & 0xFF
            pixels[idx + 2] = 128
    with open(path, "wb") as f:
        f.write(header)
        f.write(pixels)


def make_sample_audio(path, seconds=2, rate=22050):
    with wave.open(path, "wb") as w:
        w.setnchannels(1)
        w.setsampwidth(2)
        w.setframerate(rate)
        frames = bytearray()
        for i in range(rate * seconds):
            frames += struct.pack("<h", int(10000 * math.sin(2 * math.pi * 440 * i / rate)))
        w.writeframes(frames)


def cmd_demo(args):
    os.makedirs("demo", exist_ok=True)
    img = "demo/sample.bmp"
    aud = "demo/sample.wav"
    make_sample_image(img)
    make_sample_audio(aud)
    text = "Hello from RC4! سلام"
    pw = args.key
    ct = rc4_encrypt(text, pw)
    img_hide(img, "demo/secret.bmp", ct)
    print("image recovered:", rc4_decrypt(img_extract("demo/secret.bmp"), pw).decode(errors="replace"))
    aud_hide(aud, "demo/secret.wav", ct)
    print("audio recovered:", rc4_decrypt(aud_extract("demo/secret.wav"), pw).decode(errors="replace"))


def main():
    p = argparse.ArgumentParser(description="RC4 (manual) + stego")
    p.add_argument("--key", default="my-secret-key")
    sub = p.add_subparsers(dest="cmd", required=True)
    d = sub.add_parser("demo")
    d.set_defaults(func=cmd_demo)
    args = p.parse_args()
    args.func(args)


if __name__ == "__main__":
    main()

نسخة RC4-drop (إسقاط البايتات الأولى لتجاوز ثغرات الانحياز الإحصائي)

اقترح الباحثان Paul & Maitra تقنية RC4-drop[n] لحل مشكلة الانحياز الإحصائي في البايتات الأولى عبر إهمال أول 768 أو 3072 بايت من تيار المفاتيح قبل البدء في التشفير الفعلي للبيانات:

def _prga_drop(S, n, drop=768): i = j = 0 for _ in range(drop): i = (i + 1) % 256 j = (j + S[i]) % 256 S[i], S[j] = S[j], S[i] إسقاط البايتات الأولى (Dropping Keystream): تشغيل الخوارزمية لـ 768 دورة وإهمال جميع المخرجات، مما يؤدي إلى خلط عناصر المصفوفة $S$ بعمق شديد ومحو أي أثر لارتباطات المفتاح الأولي.

pt_drop = rc4_drop_decrypt(ct_drop, "DropKey2026", drop=768)
print("Decrypted Plaintext:", pt_drop.decode())

from Crypto.Cipher import ARC4 استيراد محرك التشفير التدفقي: استيراد وحدة ARC4 (Alleged RC4) المعتمدة دولياً والمكتوبة بلغة C فائقة الأداء من مكتبة PyCryptodome.
cipher = ARC4.new(key) تهيئة كائن التشفير: تزويد المحرك بالمفتاح ليقوم تلقائياً بتنفيذ خوارزمية KSA وتجهيز جدول الحالة الداخلي للبدء في إنتاج التيار.
return cipher.encrypt(data) التشفير بدون حشو: توليد بايتات التيار وتطبيق عملية XOR مباشرة على البيانات دون الحاجة لأي حشو (Padding)، وهي ميزة حصرية للتشفير التدفقي.
return cipher.decrypt(ciphertext) فك التشفير: تمرير النص المشفر على كائن جديد مهيأ بنفس المفتاح لإعادة توليد نفس تيار المفاتيح واسترجاع النص الأصلي.

"""RC4 using ready-made library (pycryptodome) + steganography."""

import argparse
import hashlib
import math
import os
import struct
import wave

from Crypto.Cipher import ARC4

from stego_image import hide as img_hide, extract as img_extract
from stego_audio import hide as aud_hide, extract as aud_extract


def derive_key(password):
    return hashlib.sha256(password.encode()).digest()


def rc4_encrypt(plaintext, password):
    key = derive_key(password)
    data = plaintext.encode() if isinstance(plaintext, str) else plaintext
    cipher = ARC4.new(key)
    return cipher.encrypt(data)


def rc4_decrypt(ciphertext, password):
    key = derive_key(password)
    cipher = ARC4.new(key)
    return cipher.decrypt(ciphertext)
